$ cat run_scenario.py
import asyncio
import langwatch, scenario

@langwatch.connect_agent(name="echo-agent")
def agent(messages) -> str:
    return f"answered {messages[-1]['content']}"

async def main():
    result = await scenario.run(
        name="connected agent under langwatch 2.0.0",
        description="the user says hello and the connected agent answers",
        agents=[agent, scenario.UserSimulatorAgent(model="openai/gpt-5-mini")],
        script=[scenario.user("hello"), scenario.agent(), scenario.succeed()],
    )
    print("success:", result.success)
    for msg in result.messages:
        print(f"  {msg['role']}: {msg['content']}")

asyncio.run(main())
